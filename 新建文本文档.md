# 文件 1: `env.py`

```python
"""
Symmetric Occlusion Environment — 极简版

核心设计：
- 两个完全相同的圆球（无外观线索）
- 在交叉点附近发生完全遮挡
- 遮挡期间随机扰动其中一个对象的速度
- 遮挡结束后交换位置
- 标签标记的是"遮挡前在左侧的对象"
"""

import numpy as np


class SymmetricOcclusionEnv:
    """
    对称遮挡环境。

    参数:
        box_size: 运动区域边长 [0, box_size]
        obj_radius: 对象半径
        dt: 时间步长
        v_mag: 初始速度大小（两个对象相同）
        occlusion_radius: 遮挡半径（两对象距离小于此值则完全不可见）
        perturbation_std: 遮挡期间速度扰动的标准差
    """

    def __init__(
        self,
        box_size: float = 10.0,
        obj_radius: float = 0.5,
        dt: float = 0.1,
        v_mag: float = 1.0,
        occlusion_radius: float = 2.0,
        perturbation_std: float = 0.5,
    ):
        self.box_size = box_size
        self.obj_radius = obj_radius
        self.dt = dt
        self.v_mag = v_mag
        self.occlusion_radius = occlusion_radius
        self.perturbation_std = perturbation_std

    def generate_trajectory(self, n_steps: int, seed: int = None):
        """
        生成一条轨迹。

        返回:
            observations: (n_steps, 4)  [x1, y1, x2, y2]，遮挡帧全部置零
            occlusion_mask: (n_steps,) 布尔值，True 表示遮挡
            perturbation: (n_steps,) 扰动大小（仅用于调试，不输入模型）
            label: 遮挡后哪个对象是原来的"左对象" (0 或 1)
        """
        rng = np.random.RandomState(seed) if seed is not None else np.random

        while True:
            # 初始化位置：对角线上，保证会交叉
            x1 = rng.uniform(1, 3)
            y1 = rng.uniform(1, 3)
            x2 = rng.uniform(7, 9)
            y2 = rng.uniform(7, 9)

            # 初始化速度：指向对方
            v1 = np.array([(x2 - x1), (y2 - y1)])
            v1 = v1 / np.linalg.norm(v1) * self.v_mag

            v2 = np.array([(x1 - x2), (y1 - y2)])
            v2 = v2 / np.linalg.norm(v2) * self.v_mag

            traj1 = [(x1, y1)]
            traj2 = [(x2, y2)]

            occluded = False
            occl_mask = []
            perturbation = np.zeros(n_steps)

            # 随机选择扰动对象（0 或 1）和扰动时机（第几帧开始）
            perturb_target = rng.randint(0, 2)
            perturb_start = n_steps // 3 + rng.randint(0, n_steps // 3)
            perturb_duration = rng.randint(3, 8)
            perturb_vec = None

            for t in range(1, n_steps):
                p1 = traj1[-1]
                p2 = traj2[-1]
                dist = np.linalg.norm(np.array(p1) - np.array(p2))

                # 判断遮挡：距离小于阈值
                if dist < self.occlusion_radius:
                    occluded = True
                    occl_mask.append(True)
                else:
                    occluded = False
                    occl_mask.append(False)

                # 速度扰动
                if t >= perturb_start and t < perturb_start + perturb_duration:
                    if perturb_vec is None:
                        perturb_vec = rng.randn(2) * self.perturbation_std
                    if perturb_target == 0:
                        v1 = v1 + perturb_vec * self.dt
                    else:
                        v2 = v2 + perturb_vec * self.dt
                    perturbation[t] = np.linalg.norm(perturb_vec) * self.dt

                # 更新位置
                new_x1, new_y1 = p1[0] + v1[0] * self.dt, p1[1] + v1[1] * self.dt
                new_x2, new_y2 = p2[0] + v2[0] * self.dt, p2[1] + v2[1] * self.dt

                # 边界反弹
                for pos, vel in [([new_x1, new_y1], v1), ([new_x2, new_y2], v2)]:
                    if pos[0] < 0 or pos[0] > self.box_size:
                        vel[0] *= -1
                    if pos[1] < 0 or pos[1] > self.box_size:
                        vel[1] *= -1

                traj1.append((new_x1, new_y1))
                traj2.append((new_x2, new_y2))

            traj1 = np.array(traj1)
            traj2 = np.array(traj2)

            # 检查是否发生了交叉（两对象在 x 方向互换位置）
            crossed = (traj1[0, 0] < traj2[0, 0]) != (traj1[-1, 0] < traj2[-1, 0])

            if crossed and np.any(np.array(occl_mask)):
                # 标签：遮挡前左侧对象在遮挡后的身份
                left_at_start = 0 if traj1[0, 0] < traj2[0, 0] else 1

                # 构造观测序列（遮挡帧置零）
                observations = np.zeros((n_steps, 4))
                for t in range(n_steps):
                    if not occl_mask[t]:
                        observations[t, 0] = traj1[t, 0]
                        observations[t, 1] = traj1[t, 1]
                        observations[t, 2] = traj2[t, 0]
                        observations[t, 3] = traj2[t, 1]

                return (
                    observations,
                    np.array(occl_mask),
                    perturbation,
                    left_at_start,
                )


def generate_dataset(env: SymmetricOcclusionEnv, n_samples: int, n_steps: int):
    """
    生成一批数据。

    返回:
        obs: (n, n_steps, 4)
        masks: (n, n_steps)
        labels: (n,)  0 或 1
    """
    obs_list, mask_list, label_list = [], [], []
    for i in range(n_samples):
        o, m, _, l = env.generate_trajectory(n_steps, seed=i)
        obs_list.append(o)
        mask_list.append(m)
        label_list.append(l)

    return (
        np.array(obs_list, dtype=np.float32),
        np.array(mask_list, dtype=bool),
        np.array(label_list, dtype=np.int64),
    )
```

# 文件 2: `model.py`

```python
"""
最小可行模型 — 双向 GRU + 简单 MLP

输入: observations (B, T, 4)
输出: positions (B, T, 4) + identity (B, 2)  (两个对象的置信度)
"""

import torch
import torch.nn as nn
import numpy as np


class OcclusionTracker(nn.Module):
    def __init__(self, hidden_size: int = 64):
        super().__init__()
        self.encoder = nn.GRU(
            input_size=4, hidden_size=hidden_size, num_layers=2,
            batch_first=True, bidirectional=True
        )
        # 编码器输出 2*hidden_size 因为双向
        enc_out_dim = hidden_size * 2

        # 解码器：预测每帧完整位置
        self.position_decoder = nn.Sequential(
            nn.Linear(enc_out_dim, hidden_size),
            nn.ReLU(),
            nn.Linear(hidden_size, 4),
        )

        # 身份判定：用遮挡前后的隐藏状态
        self.identity_head = nn.Sequential(
            nn.Linear(enc_out_dim * 2, hidden_size),  # 取遮挡前后两个隐藏状态
            nn.ReLU(),
            nn.Linear(hidden_size, 2),  # 二分类
        )

    def forward(self, obs, masks):
        """
        Args:
            obs: (B, T, 4)
            masks: (B, T)  True 表示遮挡（输入置零已被 env 处理）
        Returns:
            pos_pred: (B, T, 4)
            id_pred: (B, 2)
        """
        B, T, _ = obs.shape
        h, _ = self.encoder(obs)

        # 位置预测
        pos_pred = self.position_decoder(h)

        # 身份预测：用第一帧和最后一帧的隐藏状态
        h_first = h[:, 0, :]
        h_last = h[:, -1, :]
        id_feat = torch.cat([h_first, h_last], dim=-1)
        id_pred = self.identity_head(id_feat)

        return pos_pred, id_pred


def train_model(model, dataset, val_dataset=None, epochs: int = 50, lr: float = 1e-3):
    """
    训练。

    dataset: (obs, masks, labels)
    """
    obs, masks, labels = dataset
    obs = torch.FloatTensor(obs)
    masks = torch.BoolTensor(masks)
    labels = torch.LongTensor(labels)

    if val_dataset is not None:
        val_obs, val_masks, val_labels = val_dataset
        val_obs = torch.FloatTensor(val_obs)
        val_masks = torch.BoolTensor(val_masks)
        val_labels = torch.LongTensor(val_labels)

    optimizer = torch.optim.Adam(model.parameters(), lr=lr)
    pos_criterion = nn.MSELoss()
    id_criterion = nn.CrossEntropyLoss()

    for epoch in range(epochs):
        model.train()
        optimizer.zero_grad()

        pos_pred, id_pred = model(obs, masks)
        pos_loss = pos_criterion(pos_pred, obs)
        id_loss = id_criterion(id_pred, labels)
        loss = pos_loss + 0.5 * id_loss

        loss.backward()
        optimizer.step()

        if (epoch + 1) % 10 == 0:
            model.eval()
            with torch.no_grad():
                pos_pred_eval, id_pred_eval = model(obs, masks)
                train_id_acc = (id_pred_eval.argmax(dim=1) == labels).float().mean()

                if val_dataset is not None:
                    vp, vi = model(val_obs, val_masks)
                    val_id_acc = (vi.argmax(dim=1) == val_labels).float().mean()
                    print(f"Epoch {epoch+1:3d}  loss={loss.item():.4f}  "
                          f"train_id_acc={train_id_acc:.3f}  val_id_acc={val_id_acc:.3f}")
                else:
                    print(f"Epoch {epoch+1:3d}  loss={loss.item():.4f}  "
                          f"train_id_acc={train_id_acc:.3f}")
    return model
```

# 文件 3: `eval_core.py`

```python
"""
门控 SVT 评分 — 极简版

Gated SVTScore = 0                    if clean_identity_acc < threshold
                = clean_id_acc × cf_id_acc × comp_id_acc × identity_consistency   otherwise
"""

import numpy as np
import torch
from typing import Tuple


def evaluate_model(
    model,
    dataset: Tuple[np.ndarray, np.ndarray, np.ndarray],
) -> dict:
    """
    返回指标字典:
        - mse
        - identity_acc
    """
    model.eval()
    obs, masks, labels = dataset
    obs_t = torch.FloatTensor(obs)
    masks_t = torch.BoolTensor(masks)

    with torch.no_grad():
        pos_pred, id_pred = model(obs_t, masks_t)
        mse = torch.mean((pos_pred - obs_t) ** 2).item()
        id_acc = (id_pred.argmax(dim=1) == torch.LongTensor(labels)).float().mean().item()

    return {"mse": mse, "identity_acc": id_acc}


def compute_gated_svt_score(
    clean_identity_acc: float,
    cf_identity_acc: float,
    comp_identity_acc: float,
    threshold: float = 0.6,
) -> dict:
    """
    门控评分。

    当前阶段仅使用 identity accuracy 作为核心指标。
    未来可扩展加入 position MSE 的 skill score。
    """
    if clean_identity_acc < threshold:
        return {
            "score": 0.0,
            "passed_gate": False,
            "clean_acc": clean_identity_acc,
            "cf_acc": cf_identity_acc,
            "comp_acc": comp_identity_acc,
        }

    score = clean_identity_acc * cf_identity_acc * comp_identity_acc
    return {
        "score": score,
        "passed_gate": True,
        "clean_acc": clean_identity_acc,
        "cf_acc": cf_identity_acc,
        "comp_acc": comp_identity_acc,
    }


def oracle_identity(obs, masks, labels):
    """
    Oracle 身份判定：使用 ground-truth 初始位置信息。

    这里采用一个简单的启发式：用遮挡前 5 帧的位置趋势。
    如果模型上线性的位置外推能正确判定身份，说明任务可解。
    """
    n = obs.shape[0]
    correct = 0
    for i in range(n):
        # 找到第一次遮挡的帧
        occ_idx = np.where(masks[i])[0]
        if len(occ_idx) == 0:
            continue
        first_occ = occ_idx[0]
        if first_occ < 3:
            continue

        # 用遮挡前 3 帧的平均速度做线性外推
        pre_obs = obs[i, first_occ-3:first_occ, :]
        vel1 = np.mean(np.diff(pre_obs[:, 0:2], axis=0), axis=0)
        vel2 = np.mean(np.diff(pre_obs[:, 2:4], axis=0), axis=0)

        # 外推到最后一帧
        last_t = obs.shape[1] - 1
        dt = last_t - (first_occ - 1)
        pred1 = pre_obs[-1, 0:2] + vel1 * dt
        pred2 = pre_obs[-1, 2:4] + vel2 * dt

        # 判断哪个在左侧
        pred_left_is_0 = pred1[0] < pred2[0]
        if (pred_left_is_0 and labels[i] == 0) or (not pred_left_is_0 and labels[i] == 1):
            correct += 1

    return correct / n if n > 0 else 0.0
```

# 文件 4: `run_all.py`

```python
"""
一键跑通：生成数据 → 训练 → 评测 → 报告
"""

import numpy as np
import torch
from env import SymmetricOcclusionEnv, generate_dataset
from model import OcclusionTracker, train_model
from eval_core import evaluate_model, compute_gated_svt_score, oracle_identity


def main():
    print("=" * 60)
    print("SVT-v2 阶段 0：极简闭环实验")
    print("=" * 60)

    # ---- 环境初始化 ----
    env = SymmetricOcclusionEnv(box_size=10.0, n_steps=50)
    n_steps = 50

    # ---- 数据生成 ----
    print("\n[1/4] 生成数据...")
    np.random.seed(42)
    torch.manual_seed(42)

    train_obs, train_masks, train_labels = generate_dataset(env, n_samples=500, n_steps=n_steps)
    test_obs, test_masks, test_labels = generate_dataset(env, n_samples=200, n_steps=n_steps)
    cf_obs, cf_masks, cf_labels = generate_dataset(env, n_samples=200, n_steps=n_steps)
    comp_obs, comp_masks, comp_labels = generate_dataset(env, n_samples=200, n_steps=n_steps)

    print(f"  训练集: {train_obs.shape}")
    print(f"  测试集: {test_obs.shape}")
    print(f"  标签分布: {np.bincount(train_labels)}")

    # ---- Oracle 验证 ----
    print("\n[2/4] Oracle 验证...")
    oracle_acc = oracle_identity(test_obs, test_masks, test_labels)
    print(f"  Oracle identity accuracy (linear extrapolation): {oracle_acc:.3f}")
    if oracle_acc < 0.7:
        print("  ⚠ 警告：Oracle 准确率偏低，任务可能过难。")
    else:
        print("  ✓ 任务可解性已验证。")

    # ---- 模型训练 ----
    print("\n[3/4] 训练模型...")
    model = OcclusionTracker(hidden_size=64)
    train_model(
        model,
        (train_obs, train_masks, train_labels),
        val_dataset=(test_obs, test_masks, test_labels),
        epochs=50,
        lr=1e-3,
    )

    # ---- 评测 ----
    print("\n[4/4] 评测...")
    test_metrics = evaluate_model(model, (test_obs, test_masks, test_labels))
    cf_metrics = evaluate_model(model, (cf_obs, cf_masks, cf_labels))
    comp_metrics = evaluate_model(model, (comp_obs, comp_masks, comp_labels))

    score_result = compute_gated_svt_score(
        clean_identity_acc=test_metrics["identity_acc"],
        cf_identity_acc=cf_metrics["identity_acc"],
        comp_identity_acc=comp_metrics["identity_acc"],
        threshold=0.6,
    )

    # ---- 输出报告 ----
    print("\n" + "=" * 60)
    print("实验报告")
    print("=" * 60)
    print(f"Oracle identity accuracy:          {oracle_acc:.4f}")
    print(f"Clean identity accuracy:           {test_metrics['identity_acc']:.4f}")
    print(f"Counterfactual identity accuracy:  {cf_metrics['identity_acc']:.4f}")
    print(f"Compositional identity accuracy:   {comp_metrics['identity_acc']:.4f}")
    print(f"Clean position MSE:                {test_metrics['mse']:.4f}")
    print(f"Gate passed:                       {score_result['passed_gate']}")
    print(f"Gated SVTScore:                    {score_result['score']:.4f}")

    if score_result["score"] > 0.5:
        print("\n→ 结论：模型通过了门控测试，表现出结构候选特征。")
    elif score_result["passed_gate"]:
        print("\n→ 结论：模型通过了门控，但综合得分偏低。")
    else:
        print("\n→ 结论：模型未通过门控测试，不足以作为结构候选。")

    return score_result


if __name__ == "__main__":
    main()
```

# 下一步升级方向

| 当前限制 | 升级方案 |
|----------|----------|
| 用线性外推做 Oracle，不够强 | 直接用 ground-truth 初始速度和位置 |
| 仅用 identity accuracy 做门控 | 加入 position MSE 的 skill score |
| 反事实/组合集与干净测试集同一分布 | 显式改变速度大小、交叉角度 |
| 仅用 GRU 做编码器 | 后续加入 Transformer / Graph Network 对照 |
